"""Remove RFR's dead fusion_align layer so DDP sees no unused parameters.

Usage on the training server:
    python fix_rfr_ddp_unused_parameters.py /path/to/RFR-main/codes/model/RFR_framework.py

fusion_align is declared by the upstream implementation but never called by
forward_train() or forward_test().  Removing it changes no network computation;
it only removes the two permanently-unused trainable tensors that break DDP.
"""

from __future__ import annotations

import re
import shutil
import sys
from pathlib import Path


def main() -> None:
    if len(sys.argv) != 2:
        raise SystemExit("Usage: python fix_rfr_ddp_unused_parameters.py /path/to/RFR_framework.py")
    path = Path(sys.argv[1]).resolve()
    source = path.read_text(encoding="utf-8")
    # Upstream formatting wraps this Conv2d declaration across two lines.
    pattern = r"^[ \t]*self\.fusion_align\s*=\s*nn\.Conv2d\((?:.|\n)*?\)\s*\n"
    updated, replacements = re.subn(pattern, "", source, count=1, flags=re.MULTILINE)
    if replacements == 0:
        if "self.fusion_align" not in source:
            print("No change: fusion_align is already absent.")
            return
        raise RuntimeError("fusion_align was found but its Conv2d declaration could not be located; patch it manually.")
    if "self.fusion_align" in updated:
        raise RuntimeError("Refusing to patch: fusion_align is referenced elsewhere in this source file.")
    backup = path.with_suffix(path.suffix + ".bak")
    if not backup.exists():
        shutil.copy2(path, backup)
    path.write_text(updated, encoding="utf-8")
    print(f"Removed unused fusion_align from {path}; backup: {backup}")


if __name__ == "__main__":
    main()
