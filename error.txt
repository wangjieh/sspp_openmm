"""Coordinate Attention neck for MMSegmentation.

Recommended usage:

    DINOv3 + Adapter
            ↓
    CoordinateAttentionNeck
            ↓
    HyPiPixelDecoder
            ↓
    Mask2Former

For tune50, the recommended first experiment is:
    CA on F1/F2 only
    F3/F4 pass through unchanged.
"""

from __future__ import annotations

from typing import Sequence, Tuple

import torch
from torch import Tensor, nn

from mmengine.model import BaseModule
from mmseg.registry import MODELS


class CoordinateAttention(nn.Module):
    """Coordinate Attention for a single 2-D feature map.

    Input:
        x: [B, C, H, W]

    Output:
        out: [B, C, H, W]

    The module separately encodes height-direction and width-direction
    information, then generates two coordinate-aware attention maps.

    Args:
        channels (int):
            Number of input/output channels.

        reduction (int):
            Channel reduction ratio.

        min_channels (int):
            Minimum hidden channel number.
    """

    def __init__(
        self,
        channels: int,
        reduction: int = 32,
        min_channels: int = 8,
    ) -> None:
        super().__init__()

        if channels < 1:
            raise ValueError(
                f'channels must be positive, got {channels}.'
            )

        if reduction < 1:
            raise ValueError(
                f'reduction must be positive, got {reduction}.'
            )

        if min_channels < 1:
            raise ValueError(
                f'min_channels must be positive, got {min_channels}.'
            )

        # ------------------------------------------------------------
        # Bottleneck channels
        #
        # tune50:
        #   C = 1024
        #   reduction = 32
        #
        # therefore:
        #   hidden_channels = 32
        # ------------------------------------------------------------
        hidden_channels = max(
            min_channels,
            channels // reduction
        )

        # ------------------------------------------------------------
        # Shared 1x1 convolution
        #
        # C -> C_mid
        # ------------------------------------------------------------
        self.conv1 = nn.Conv2d(
            in_channels=channels,
            out_channels=hidden_channels,
            kernel_size=1,
            stride=1,
            padding=0,
            bias=False,
        )

        self.bn1 = nn.BatchNorm2d(
            hidden_channels
        )

        # Coordinate Attention originally uses h-swish.
        self.act = nn.Hardswish(
            inplace=True
        )

        # ------------------------------------------------------------
        # Height-direction attention
        #
        # C_mid -> C
        # ------------------------------------------------------------
        self.conv_h = nn.Conv2d(
            in_channels=hidden_channels,
            out_channels=channels,
            kernel_size=1,
            stride=1,
            padding=0,
            bias=True,
        )

        # ------------------------------------------------------------
        # Width-direction attention
        #
        # C_mid -> C
        # ------------------------------------------------------------
        self.conv_w = nn.Conv2d(
            in_channels=hidden_channels,
            out_channels=channels,
            kernel_size=1,
            stride=1,
            padding=0,
            bias=True,
        )

    def forward(
        self,
        x: Tensor,
    ) -> Tensor:

        if x.ndim != 4:
            raise ValueError(
                'CoordinateAttention expects input shape '
                '[B, C, H, W], '
                f'but got {tuple(x.shape)}.'
            )

        _, _, h, w = x.shape

        identity = x

        # ============================================================
        # Height branch
        # ============================================================
        #
        # Average pooling along width:
        #
        # [B, C, H, W]
        #       ↓
        # [B, C, H, 1]
        #
        # Spatial information along H is preserved.
        # ============================================================
        x_h = x.mean(
            dim=3,
            keepdim=True
        )

        # ============================================================
        # Width branch
        # ============================================================
        #
        # Average pooling along height:
        #
        # [B, C, H, W]
        #       ↓
        # [B, C, 1, W]
        #
        # Then transpose:
        #
        # [B, C, W, 1]
        #
        # so it can be concatenated with x_h.
        # ============================================================
        x_w = x.mean(
            dim=2,
            keepdim=True
        )

        x_w = x_w.permute(
            0, 1, 3, 2
        )

        # ============================================================
        # Coordinate information fusion
        # ============================================================
        #
        # x_h: [B, C, H, 1]
        # x_w: [B, C, W, 1]
        #
        # concatenate:
        #
        # [B, C, H+W, 1]
        # ============================================================
        y = torch.cat(
            (x_h, x_w),
            dim=2
        )

        # ------------------------------------------------------------
        # Shared channel compression
        #
        # [B, C, H+W, 1]
        #       ↓
        # [B, C_mid, H+W, 1]
        # ------------------------------------------------------------
        y = self.conv1(y)
        y = self.bn1(y)
        y = self.act(y)

        # ============================================================
        # Split H / W branches
        # ============================================================
        y_h, y_w = torch.split(
            y,
            [h, w],
            dim=2
        )

        # y_h:
        # [B, C_mid, H, 1]

        # y_w:
        # [B, C_mid, W, 1]
        #       ↓
        # [B, C_mid, 1, W]
        y_w = y_w.permute(
            0, 1, 3, 2
        )

        # ============================================================
        # Generate coordinate attention maps
        # ============================================================

        # Height attention:
        #
        # [B, C_mid, H, 1]
        #       ↓
        # [B, C, H, 1]
        a_h = self.conv_h(y_h)
        a_h = torch.sigmoid(a_h)

        # Width attention:
        #
        # [B, C_mid, 1, W]
        #       ↓
        # [B, C, 1, W]
        a_w = self.conv_w(y_w)
        a_w = torch.sigmoid(a_w)

        # ============================================================
        # Coordinate Attention
        # ============================================================
        #
        # Broadcasting:
        #
        # x:
        #     [B, C, H, W]
        #
        # a_h:
        #     [B, C, H, 1]
        #
        # a_w:
        #     [B, C, 1, W]
        #
        # result:
        #
        #     [B, C, H, W]
        # ============================================================
        out = identity * a_h * a_w

        return out


class _CAEnhanceBlock(nn.Module):
    """Coordinate Attention with optional residual stabilization.

    Standard CA:

        F_ca = CA(F)

    When residual=True:

        F_out = F + gamma * (F_ca - F)

    This formulation is deliberately used for tune50 because DINOv3 and
    Adapter already contain pretrained / well-structured representations.

    With a small gamma_init, the network initially behaves almost exactly
    like the original baseline:

        gamma ~= 0
            =>
        F_out ~= F

    but CA still receives gradients because gamma is initialized to a
    small non-zero value rather than exactly zero.
    """

    def __init__(
        self,
        channels: int,
        reduction: int = 32,
        min_channels: int = 8,
        residual: bool = True,
        gamma_init: float = 1e-3,
    ) -> None:
        super().__init__()

        self.ca = CoordinateAttention(
            channels=channels,
            reduction=reduction,
            min_channels=min_channels,
        )

        self.residual = residual

        if residual:

            # One learnable scalar for one feature level.
            #
            # F1 has gamma_1
            # F2 has gamma_2
            #
            # They can independently learn how strongly CA should
            # participate in feature enhancement.
            self.gamma = nn.Parameter(
                torch.tensor(
                    float(gamma_init),
                    dtype=torch.float32
                )
            )

        else:
            self.register_parameter(
                'gamma',
                None
            )

    def forward(
        self,
        x: Tensor,
    ) -> Tensor:

        attended = self.ca(x)

        # ------------------------------------------------------------
        # Standard Coordinate Attention
        # ------------------------------------------------------------
        if not self.residual:
            return attended

        # ------------------------------------------------------------
        # Residual Coordinate Attention
        #
        # out = x + gamma * (CA(x) - x)
        #
        # gamma = 0:
        #     exact identity
        #
        # gamma = 1:
        #     standard CA
        # ------------------------------------------------------------
        out = x + self.gamma * (
            attended - x
        )

        return out


@MODELS.register_module()
class CoordinateAttentionNeck(BaseModule):
    """Coordinate Attention Neck for multi-scale MMSeg features.

    The neck:

        1. does NOT perform channel projection;
        2. does NOT perform FPN fusion;
        3. does NOT change feature resolution;
        4. only performs coordinate-aware feature recalibration.

    Therefore:

        input:
            F1, F2, F3, F4

        output:
            F1', F2', F3', F4'

    have exactly the same shapes.

    Recommended tune50 configuration:

        in_channels=[
            1024,
            1024,
            1024,
            1024
        ]

        apply_levels=(0, 1)

    which means:

        F1 -> CA
        F2 -> CA
        F3 -> Identity
        F4 -> Identity

    Args:
        in_channels:
            Channel numbers of all input feature levels.

        apply_levels:
            Zero-based feature indices where CA is enabled.

            (0,)       -> F1
            (0, 1)     -> F1 + F2
            (0, 1, 2)  -> F1 + F2 + F3
            (0, 1, 2, 3)
                       -> all levels

        reduction:
            Coordinate Attention reduction ratio.

        min_channels:
            Minimum hidden channel number.

        residual:
            Whether to use residual stabilization.

        gamma_init:
            Initial value of learnable gamma.

        init_cfg:
            MMEngine initialization configuration.
    """

    def __init__(
        self,
        in_channels: Sequence[int],
        apply_levels: Sequence[int] = (0, 1),
        reduction: int = 32,
        min_channels: int = 8,
        residual: bool = True,
        gamma_init: float = 1e-3,
        init_cfg=None,
    ) -> None:

        super().__init__(
            init_cfg=init_cfg
        )

        # ============================================================
        # Basic checks
        # ============================================================
        if (
            not isinstance(
                in_channels,
                Sequence
            )
            or len(in_channels) == 0
        ):
            raise ValueError(
                'in_channels must be a non-empty sequence.'
            )

        if any(
            int(c) < 1
            for c in in_channels
        ):
            raise ValueError(
                'All values in in_channels must be positive, '
                f'got {in_channels}.'
            )

        if reduction < 1:
            raise ValueError(
                f'reduction must be positive, got {reduction}.'
            )

        if min_channels < 1:
            raise ValueError(
                'min_channels must be positive, '
                f'got {min_channels}.'
            )

        self.in_channels = tuple(
            int(c)
            for c in in_channels
        )

        # CA does not modify channel dimensions.
        self.out_channels = self.in_channels

        self.num_levels = len(
            self.in_channels
        )

        # ============================================================
        # Check apply_levels
        # ============================================================
        levels = tuple(
            int(i)
            for i in apply_levels
        )

        if len(set(levels)) != len(levels):
            raise ValueError(
                'apply_levels contains duplicate indices: '
                f'{apply_levels}.'
            )

        for level in levels:

            if (
                level < 0
                or level >= self.num_levels
            ):
                raise ValueError(
                    'apply_levels contains invalid level '
                    f'{level}. '
                    'Valid range is '
                    f'[0, {self.num_levels - 1}].'
                )

        self.apply_levels = tuple(
            sorted(levels)
        )

        enabled_levels = set(
            self.apply_levels
        )

        # ============================================================
        # Build CA / Identity blocks
        # ============================================================
        blocks = []

        for level, channels in enumerate(
            self.in_channels
        ):

            # --------------------------------------------------------
            # Apply CA
            # --------------------------------------------------------
            if level in enabled_levels:

                block = _CAEnhanceBlock(
                    channels=channels,
                    reduction=reduction,
                    min_channels=min_channels,
                    residual=residual,
                    gamma_init=gamma_init,
                )

            # --------------------------------------------------------
            # Pass feature through unchanged
            # --------------------------------------------------------
            else:

                block = nn.Identity()

            blocks.append(block)

        self.blocks = nn.ModuleList(
            blocks
        )

    def forward(
        self,
        inputs: Sequence[Tensor],
    ) -> Tuple[Tensor, ...]:

        # ============================================================
        # Check number of feature levels
        # ============================================================
        if len(inputs) != self.num_levels:

            raise RuntimeError(
                'CoordinateAttentionNeck received an unexpected '
                'number of feature levels: '
                f'expected {self.num_levels}, '
                f'got {len(inputs)}.'
            )

        outputs = []

        # ============================================================
        # F1 / F2 / F3 / F4
        # ============================================================
        for level, (
            feature,
            expected_channels,
            block,
        ) in enumerate(
            zip(
                inputs,
                self.in_channels,
                self.blocks
            )
        ):

            # --------------------------------------------------------
            # Feature must be:
            #
            # [B, C, H, W]
            # --------------------------------------------------------
            if feature.ndim != 4:

                raise RuntimeError(
                    f'Feature F{level + 1} must be a '
                    '4-D tensor [B, C, H, W], '
                    f'got shape {tuple(feature.shape)}.'
                )

            # --------------------------------------------------------
            # Check channels
            # --------------------------------------------------------
            if (
                feature.shape[1]
                != expected_channels
            ):

                raise RuntimeError(
                    f'Feature F{level + 1} channel mismatch: '
                    f'expected {expected_channels}, '
                    f'got {feature.shape[1]}.'
                )

            # --------------------------------------------------------
            # CA or Identity
            # --------------------------------------------------------
            feature = block(feature)

            outputs.append(feature)

        return tuple(outputs)
